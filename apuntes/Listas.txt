#include <iostream>
#include <conio.h>
using namespace std;

struct Rasgo {//esta estructura representa un rasgo de una persona, con un valor y punteros para formar una lista doblemente enlazada de rasgos.
    string valor;
    Rasgo *next;
    Rasgo *prior;
};

struct nodo {//esta estructura representa una persona, con un id, nombre, punteros a la cabeza y cola de su lista de rasgos, y punteros para formar una lista doblemente enlazada de personas.
    int id;
    string nombre;
    Rasgo *rasgosHead;
    Rasgo *rasgosTail;
    nodo *next;
    nodo *prior;
};

void liberarRasgos(nodo *persona) {//esta función libera la memoria de los rasgos asociados a una persona, eliminando cada nodo de rasgo y estableciendo los punteros de cabeza y cola a NULL.  
    Rasgo *cur = persona->rasgosHead;
    while (cur != NULL) {
        Rasgo *next = cur->next;
        delete cur;
        cur = next;
    }
    persona->rasgosHead = persona->rasgosTail = NULL;
}

void agregarRasgoALista(nodo *persona, const string &rasgo) {//esta función agrega un nuevo rasgo a la lista de rasgos de una persona. Crea un nuevo nodo de rasgo, lo inicializa con el valor proporcionado, y lo agrega al final de la lista de rasgos de la persona, actualizando los punteros de cabeza y cola según sea necesario.
    Rasgo *nuevo = new Rasgo;
    nuevo->valor = rasgo;
    nuevo->next = NULL;
    nuevo->prior = NULL;

    if (persona->rasgosHead == NULL) {
        persona->rasgosHead = persona->rasgosTail = nuevo;
    } else {
        persona->rasgosTail->next = nuevo;
        nuevo->prior = persona->rasgosTail;
        persona->rasgosTail = nuevo;
    }
}

void ingresarRasgosParaPersona(nodo *persona) {//esta función permite al usuario ingresar rasgos para una persona específica. Solicita al usuario que ingrese rasgos uno por uno, agregándolos a la lista de rasgos de la persona utilizando la función agregarRasgoALista. El proceso continúa hasta que el usuario decida no agregar más rasgos.
    char seguir = 'n';

    do {
        string rasgo;
        cout << "Ingrese un rasgo (o deje vacio y presione Enter para terminar): ";
        getline(cin, rasgo);
        if (rasgo.empty()) {
            break;
        }
        agregarRasgoALista(persona, rasgo);

        cout << "Desea agregar otro rasgo? (s/n): ";
        cin >> seguir;
        cin.ignore();
    } while (seguir == 's' || seguir == 'S');
}

//INTERCAMBIAR CONTENIDO E INVERTIR LISTA
void intercambiarContenidoPorID(nodo *&primero) {// punto 1 /// esta función permite al usuario intercambiar el contenido de dos nodos en la lista de personas utilizando sus IDs. Solicita al usuario que ingrese los IDs de los dos nodos a intercambiar, busca los nodos correspondientes en la lista, y luego intercambia sus campos (ID, nombre y lista de rasgos) sin modificar la estructura de la lista. Además, ofrece la opción de invertir toda la lista, lo que cambia el orden de los nodos sin alterar su contenido.
    int opcion;
    cout << "\nQue desea hacer?\n";
    cout << "1. Intercambiar contenido de dos nodos por ID\n";
    cout << "2. Invertir lista\n";
    cout << "Seleccione una opcion: ";
    cin >> opcion;

    if (opcion == 1) {
        // Intercambiar contenido
        int id1, id2;
        cout << "Ingrese el primer ID: ";
        cin >> id1;
        cout << "Ingrese el segundo ID: ";
        cin >> id2;

        nodo *nodo1 = NULL;
        nodo *nodo2 = NULL;
        nodo *temp = primero;

        while (temp != NULL) {
            if (temp->id == id1)
                nodo1 = temp;
            if (temp->id == id2)
                nodo2 = temp;
            temp = temp->next;
        }

        if (nodo1 == NULL || nodo2 == NULL) {
            cout << "Uno o ambos IDs no existen\n";
            return;
        }

        swap(nodo1->id, nodo2->id);
        swap(nodo1->nombre, nodo2->nombre);
        swap(nodo1->rasgosHead, nodo2->rasgosHead);
        swap(nodo1->rasgosTail, nodo2->rasgosTail);

        cout << "Contenido intercambiado correctamente\n";
    }
    else if (opcion == 2) {
        // Invertir lista
        nodo *prev = NULL;
        nodo *curr = primero;
        nodo *next = NULL;
        while (curr != NULL) {
            next = curr->next;
            curr->next = prev;
            curr->prior = next;
            prev = curr;
            curr = next;
        }
        primero = prev;
        if (primero) primero->prior = NULL;
        cout << "Lista invertida correctamente\n";
    }
    else {
        cout << "Opcion invalida\n";
    }
}

#include <string>
#include <iostream>
#include <conio.h>
#include <limits>
using namespace std;

struct programacion{
    int franjaHoraria;
    string nombre;
    string clasificacion;
    programacion *next;
    programacion *prior;
};

struct nodo {
    int id;
    string canal;
    programacion *programacionesHead;
    programacion *programacionesTail;
    nodo *next;
    nodo *prior;
};

void liberarProgramaciones(nodo *canal) {
    programacion *cur = canal->programacionesHead;
    while (cur != NULL) {
        programacion *next = cur->next;
        delete cur;
        cur = next;
    }
    canal->programacionesHead = canal->programacionesTail = NULL;
}

void agregarProgramacionALista(nodo *canal, int franja, const string &nom, const string &clas) {
    programacion *temp = canal->programacionesHead;
    while (temp != NULL) {
        if (temp->franjaHoraria == franja) {
            cout << "Franja ocupada\n";
            return;
        }
        temp = temp->next;
    }

    programacion *nuevo = new programacion{franja, nom, clas};
    nuevo->next = nuevo->prior = NULL;

    if (canal->programacionesHead == NULL) {
        canal->programacionesHead = canal->programacionesTail = nuevo;
    } else if (franja < canal->programacionesHead->franjaHoraria) {
        nuevo->next = canal->programacionesHead;
        canal->programacionesHead->prior = nuevo;
        canal->programacionesHead = nuevo;
    } else {
        temp = canal->programacionesHead;
        while (temp->next != NULL && temp->next->franjaHoraria < franja) {
            temp = temp->next;
        }
        nuevo->next = temp->next;
        nuevo->prior = temp;
        if (temp->next != NULL) {
            temp->next->prior = nuevo;
        } else {
            canal->programacionesTail = nuevo;
        }
        temp->next = nuevo;
    }
}

void ingresarProgramacionesParaCanal(nodo *canal) {
    char seguir = 'n';

    do {
        int franja;
        string nom, clas;
        cout << "Franja horaria (0-23): ";
        cin >> franja;
        if (cin.fail() || franja < 0 || franja > 23) {

        }

        cout << "Desea agregar otra programacion? (s/n): ";
        cin >> seguir;
        cin.ignore();
    } while (seguir == 's' || seguir == 'S');
}

void agregarCanalordenado(nodo *&primero) {
    nodo *nuevo = new nodo;

    cout << "Ingrese el numero del canal: ";
    cin >> nuevo->id;
    if(cin.fail()) {
        cout << "Entrada invalida. El numero del canal debe ser un entero.\n";
        cin.clear();
        cin.ignore(numeric_limits<streamsize>::max(), '\n');
        delete nuevo;
        return;
    }
    cin.ignore();

    // Verificar si el canal ya existe
    if (primero != NULL) {
        nodo *temp = primero;
        do {
            if (temp->id == nuevo->id) {
                cout << "Canal ya existe\n";
                delete nuevo;
                return;
            }
        } while ((temp = temp->next) != primero);
    }

    cout << "Ingrese el canal: ";
    getline(cin, nuevo->canal);
    
    nuevo->programacionesHead = nuevo->programacionesTail = NULL;

    cout << "\n Opcional: agregar programaciones para este canal \n";
    ingresarProgramacionesParaCanal(nuevo);

    if (primero == NULL) {
        nuevo->next = nuevo->prior = nuevo;
        primero = nuevo;
        return;
    }
    
    nodo *temp = primero;
    while (temp->next != primero && temp->next->id > nuevo->id) 
        temp = temp->next;
    
    nuevo->next = temp->next;
    nuevo->prior = temp;
    temp->next->prior = nuevo;
    temp->next = nuevo;
    
    if (nuevo->id > primero->id) 
        primero = nuevo;
}

void eliminarnodocanalindicado(nodo *&primero) {
    nodo *canal = buscarCanal(primero);
    if (canal == NULL) {
        cout << "Canal no encontrado\n";
        return;
    }

    liberarProgramaciones(canal);

    if (canal->next == canal) {
        primero = NULL;
    } else {
        canal->prior->next = canal->next;
        canal->next->prior = canal->prior;
        if (primero == canal) {
            primero = canal->next;
        }
    }
    delete canal;
    cout << "Canal eliminado\n";

void eliminarnodoposicionindicada(nodo *&primero) {
    int pos;
    cout << "Ingrese la posicion a eliminar el nodo: ";
    cin >> pos;
    cin.ignore();

    if (primero == NULL) {
        cout << "La lista esta vacia\n";
        return;
    }

    if (pos < 1) {
        cout << "La posicion no existe\n";
        return;
    }

    nodo *temp = primero;
    int i = 1;
    
    if (pos == 1) {
        if (temp->next == primero) {
            primero = NULL;
        } else {
            nodo *ultimo = primero->prior;
            primero = temp->next;
            primero->prior = ultimo;
            ultimo->next = primero;
        }
        delete temp;
        cout << "Nodo eliminado\n";
    }
    else {
        while (i < pos - 1 && temp->next != primero) {
            temp = temp->next;
            i++;
        }

        if (temp->next == primero && i < pos - 1) {
            cout << "La posicion no existe\n";
        }
        else {
            nodo *temp2 = temp->next;
            temp->next = temp2->next;
            temp2->next->prior = temp;
            delete temp2;
            cout << "Nodo eliminado\n";
        }
    }
}

        cout << "Flechas: Izq/Der para programas, Arriba/Abajo para canales, Escape para salir\n";

        if (ch == 224) {
            ch = getch();
            if (ch == 75 && progActual != NULL && progActual->prior != NULL) { // Izq
                progActual = progActual->prior;
            } else if (ch == 77 && progActual != NULL && progActual->next != NULL) { // Der

                progActual = canalActual->programacionesHead;
            }
        } else if (ch == 27) { // Escape
            break;
        } else if (progActual == NULL && canalActual->programacionesHead != NULL) {
            progActual = canalActual->programacionesHead;
        }
    }
}

struct nodo{
    string palabra;
    nodo *next;
    nodo *prior;

void agregarpalabra(nodo *&primero) {
    nodo *nuevo = new nodo;

    cout << "Ingrese la palabra: ";
    getline(cin, nuevo->palabra);

    if (primero == NULL) {
        nuevo->next = nuevo->prior = nuevo;
        primero = nuevo;
        return;
    }
    
    nodo *temp = primero;
    while (temp->next != primero && temp->next->palabra > nuevo->palabra) 
        temp = temp->next;
    
    nuevo->next = temp->next;
    nuevo->prior = temp;
    temp->next->prior = nuevo;
    temp->next = nuevo;
    
    if (nuevo->palabra > primero->palabra) 
        primero = nuevo;
}

void separar(nodo *original, nodo *&impares, nodo *&pares) {
    if (original == NULL) return;

    nodo *temp = original;
    int pos = 1;

    do {
        nodo *nuevo = new nodo;
        nuevo->palabra = temp->palabra;

        if (pos % 2 == 1) {
            if (impares == NULL) {
                nuevo->next = nuevo;
                nuevo->prior = nuevo;
                impares = nuevo;
            } else {
                nodo *u = impares->prior;
                u->next = nuevo;
                nuevo->prior = u;
                nuevo->next = impares;
                impares->prior = nuevo;
            }
        } else {
            if (pares == NULL) {
                nuevo->next = nuevo;
                nuevo->prior = nuevo;
                pares = nuevo;
            } else {
                nodo *u = pares->prior;
                u->next = nuevo;
                nuevo->prior = u;
                nuevo->next = pares;
                pares->prior = nuevo;
            }
        }

        temp = temp->next;
        pos++;
    } while (temp != original);
}
